# wbw-security 模块使用指南

## 配置说明

### 最小配置（只需密钥）

```yaml
# 最小配置（只需密钥）
wbw:
  jwt:
    secret: your-jwt-secret-key-change-in-production
```

### 完整配置

```yaml
# 完整配置
wbw:
  jwt:
    enabled: true
    secret: your-jwt-secret-key-change-in-production
    access-token-expire: 7200        # 2小时
    refresh-token-expire: 2592000    # 30天
    header: Authorization
    prefix: "Bearer "
    context-enabled: true
    white-list:
      - "/api/public/**"
      - "/auth/**"
      - "/swagger/**"
      - "/v3/api-docs/**"
    filter:
      enabled: false  # 默认不启用过滤器（在Gateway中验证）
```

## 配置项详细说明

| 配置参数名称 | 数据类型 | 默认值 | 详细功能描述 |
| --- | --- | --- | --- |
| `wbw.jwt.enabled` | boolean | true | 是否启用JWT功能 |
| `wbw.jwt.secret` | String | 无 | JWT签名密钥，生产环境必须修改 |
| `wbw.jwt.access-token-expire` | int | 7200 | 访问令牌过期时间（秒） |
| `wbw.jwt.refresh-token-expire` | int | 2592000 | 刷新令牌过期时间（秒） |
| `wbw.jwt.header` | String | Authorization | 令牌头名称 |
| `wbw.jwt.prefix` | String | "Bearer " | 令牌前缀 |
| `wbw.jwt.context-enabled` | boolean | true | 是否启用安全上下文 |
| `wbw.jwt.white-list` | List<String> | [] | 白名单路径，不需要验证令牌 |
| `wbw.jwt.filter.enabled` | boolean | false | 是否启用JWT过滤器 |

## 不同配置组合下的行为差异

| 配置组合 | 行为差异 |
| --- | --- |
| 最小配置 | 仅启用JWT基本功能，使用默认过期时间和头信息 |
| 完整配置 | 启用所有JWT功能，可自定义所有参数 |
| `filter.enabled: true` | 在应用内部启用JWT过滤器，适用于独立应用 |
| `filter.enabled: false` | 在Gateway中验证令牌，适用于微服务架构 |
| `context-enabled: true` | 启用安全上下文，可通过JwtSecurityContext获取当前用户信息 |
| 配置白名单 | 白名单路径不需要验证令牌，直接访问 |

## 使用示例

### 1. Gateway中的JWT验证过滤器

```java
// Gateway中的JWT验证过滤器
@Component
public class JwtGatewayFilter implements GlobalFilter, Ordered {

    @Autowired
    private JwtTokenService jwtTokenService;
    
    @Autowired
    private JwtProperties jwtProperties;
    
    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        ServerHttpRequest request = exchange.getRequest();
        String path = request.getPath().value();
        
        // 检查是否为白名单路径
        if (isWhiteList(path)) {
            return chain.filter(exchange);
        }
        
        // 提取Token
        String token = extractToken(request);
        if (token == null) {
            return unauthorized(exchange, "缺少访问令牌");
        }
        
        // 验证Token
        if (!jwtTokenService.validateToken(token)) {
            return unauthorized(exchange, "访问令牌无效或已过期");
        }
        
        // 解析用户信息
        JwtUser user = jwtTokenService.parseToken(token);
        if (user == null) {
            return unauthorized(exchange, "令牌解析失败");
        }
        
        // 将用户信息添加到请求头，传递给下游服务
        ServerHttpRequest newRequest = request.mutate()
            .header("X-User-Id", user.getUserId())
            .header("X-Username", user.getUsername())
            .header("X-Tenant-Id", user.getTenantId())
            .header("X-Role", user.getRole())
            .build();
        
        return chain.filter(exchange.mutate().request(newRequest).build());
    }
    
    private boolean isWhiteList(String path) {
        // 检查白名单配置
        List<String> whiteList = jwtProperties.getWhiteList();
        return whiteList.stream().anyMatch(pattern -> path.matches(pattern.replace("**", ".*")));
    }
    
    private String extractToken(ServerHttpRequest request) {
        String authHeader = request.getHeaders().getFirst(jwtProperties.getHeader());
        if (authHeader != null && authHeader.startsWith(jwtProperties.getPrefix())) {
            return authHeader.substring(jwtProperties.getPrefix().length());
        }
        return null;
    }
    
    private Mono<Void> unauthorized(ServerWebExchange exchange, String message) {
        ServerHttpResponse response = exchange.getResponse();
        response.setStatusCode(HttpStatus.UNAUTHORIZED);
        response.getHeaders().setContentType(MediaType.APPLICATION_JSON);
        
        Map<String, Object> error = Map.of(
            "code", 401,
            "message", message,
            "timestamp", System.currentTimeMillis()
        );
        
        DataBuffer buffer = response.bufferFactory().wrap(JsonUtils.toJson(error).getBytes());
        return response.writeWith(Mono.just(buffer));
    }
    
    @Override
    public int getOrder() {
        return Ordered.HIGHEST_PRECEDENCE;
    }
}
```

### 2. 在Controller中使用

```java
// 在Controller中使用
@RestController
@RequestMapping("/api/user")
public class UserController {

    @Autowired
    private JwtTokenService jwtTokenService;
    
    /**
     * 公开接口，不需要Token
     */
    @GetMapping("/public/info")
    @Anonymous
    public Result<?> publicInfo() {
        return Result.success("这是公开信息");
    }
    
    /**
     * 受保护接口，需要Token（在Gateway中已验证）
     */
    @GetMapping("/profile")
    public Result<?> getProfile(HttpServletRequest request) {
        // 从请求头获取用户信息（由Gateway传递）
        String userId = request.getHeader("X-User-Id");
        String username = request.getHeader("X-Username");
        
        // 或者使用安全上下文（如果启用了过滤器）
        if (JwtSecurityContext.isAuthenticated()) {
            userId = JwtSecurityContext.getUserId();
            username = JwtSecurityContext.getUsername();
        }
        
        UserProfile profile = userService.getProfile(userId);
        return Result.success(profile);
    }
    
    /**
     * 生成Token接口（用于登录）
     */
    @PostMapping("/login")
    @Anonymous
    public Result<?> login(@RequestBody LoginRequest request) {
        // 验证用户名密码
        User user = authService.authenticate(request.getUsername(), request.getPassword());
        
        // 构建JWT用户信息
        JwtUser jwtUser = new JwtUser();
        jwtUser.setUserId(user.getId());
        jwtUser.setUsername(user.getUsername());
        jwtUser.setRole(user.getRole());
        jwtUser.setTenantId(user.getTenantId());
        jwtUser.setAdmin(user.isAdmin());
        jwtUser.addExtra("deptId", user.getDeptId());
        
        // 生成Token
        TokenInfo tokenInfo = jwtTokenService.generateToken(jwtUser);
        
        return Result.success(tokenInfo);
    }
    
    /**
     * 刷新Token
     */
    @PostMapping("/refresh")
    @Anonymous
    public Result<?> refreshToken(@RequestParam String refreshToken) {
        TokenInfo tokenInfo = jwtTokenService.refreshToken(refreshToken);
        return Result.success(tokenInfo);
    }
}
```

### 3. 在Service中使用

```java
// 在Service中使用
@Service
public class UserService {

    @Autowired
    private JwtTokenService jwtTokenService;
    
    @Autowired
    private UserRepository userRepository;
    
    /**
     * 获取当前用户资料
     */
    public UserProfile getCurrentUserProfile() {
        // 从安全上下文获取用户信息
        String userId = JwtSecurityContext.getUserId();
        if (userId == null) {
            throw new RuntimeException("用户未登录");
        }
        
        // 查询数据库
        User user = userRepository.findById(userId)
            .orElseThrow(() -> new RuntimeException("用户不存在"));
        
        return convertToProfile(user);
    }
    
    /**
     * 生成Token的工具方法
     */
    public TokenInfo generateUserToken(User user) {
        JwtUser jwtUser = new JwtUser();
        jwtUser.setUserId(user.getId());
        jwtUser.setUsername(user.getUsername());
        jwtUser.setRole(user.getRole());
        jwtUser.setTenantId(user.getTenantId());
        jwtUser.setAdmin(user.isAdmin());
        
        // 添加额外信息
        Map<String, Object> extra = new HashMap<>();
        extra.put("deptId", user.getDeptId());
        extra.put("email", user.getEmail());
        extra.put("phone", user.getPhone());
        jwtUser.setExtra(extra);
        
        return jwtTokenService.generateToken(jwtUser);
    }
    
    private UserProfile convertToProfile(User user) {
        // 转换逻辑
        UserProfile profile = new UserProfile();
        profile.setId(user.getId());
        profile.setUsername(user.getUsername());
        profile.setEmail(user.getEmail());
        profile.setPhone(user.getPhone());
        profile.setRole(user.getRole());
        profile.setDeptId(user.getDeptId());
        return profile;
    }
}
```

### 4. Token控制器

```java
// Token控制器
@RestController
@RequestMapping("/api/token")
public class TokenController {

    @Autowired
    private JwtTokenService jwtTokenService;
    
    /**
     * 生成Token
     */
    @PostMapping("/generate")
    @Anonymous
    public Result<?> generateToken(@RequestBody TokenRequest request) {
        try {
            TokenInfo tokenInfo = jwtTokenService.generateToken(request);
            return Result.success(tokenInfo);
        } catch (Exception e) {
            return Result.error("生成Token失败: " + e.getMessage());
        }
    }
    
    /**
     * 验证Token
     */
    @PostMapping("/validate")
    @Anonymous
    public Result<?> validateToken(@RequestParam String token) {
        boolean valid = jwtTokenService.validateToken(token);
        if (valid) {
            JwtUser user = jwtTokenService.parseToken(token);
            return Result.success(Map.of(
                "valid", true,
                "user", user
            ));
        }
        return Result.success(Map.of("valid", false));
    }
    
    /**
     * 刷新Token
     */
    @PostMapping("/refresh")
    @Anonymous
    public Result<?> refreshToken(@RequestBody RefreshRequest request) {
        try {
            TokenInfo tokenInfo = jwtTokenService.refreshToken(request.getRefreshToken());
            return Result.success(tokenInfo);
        } catch (Exception e) {
            return Result.error("刷新Token失败: " + e.getMessage());
        }
    }
    
    /**
     * 解析Token
     */
    @PostMapping("/parse")
    @Anonymous
    public Result<?> parseToken(@RequestParam String token) {
        JwtUser user = jwtTokenService.parseToken(token);
        if (user != null) {
            return Result.success(user);
        }
        return Result.error("Token解析失败");
    }
}
```

## 常见问题解决方案

### 1. Token验证失败

**症状**：JWT验证失败，返回401错误。

**解决方案**：
- 检查Token是否过期
- 检查Token格式是否正确
- 检查JWT密钥是否匹配
- 检查Token是否被篡改

### 2. 安全上下文获取用户信息为空

**症状**：`JwtSecurityContext.getUserId()` 返回null。

**解决方案**：
- 检查是否启用了安全上下文（`context-enabled: true`）
- 检查是否在Gateway中正确传递了用户信息
- 检查是否启用了过滤器（如果不使用Gateway）

### 3. 白名单路径不起作用

**症状**：配置了白名单路径，但仍然需要Token验证。

**解决方案**：
- 检查白名单路径配置格式是否正确
- 检查路径匹配规则是否正确
- 检查过滤器是否正确实现了白名单逻辑

### 4. Token过期时间设置不生效

**症状**：设置了Token过期时间，但实际过期时间与设置不符。

**解决方案**：
- 检查配置项名称是否正确
- 检查时间单位是否为秒
- 检查是否在生成Token时传递了正确的过期时间

## 最佳实践

1. **生产环境密钥管理**：使用环境变量或配置中心管理JWT密钥，避免硬编码
2. **Token过期时间设置**：根据业务场景设置合理的过期时间
3. **刷新Token机制**：实现刷新Token机制，避免用户频繁登录
4. **安全上下文使用**：优先使用安全上下文获取用户信息，简化代码
5. **白名单配置**：合理配置白名单路径，提高系统安全性
6. **Token验证位置**：在微服务架构中，建议在Gateway层统一验证Token
7. **异常处理**：实现统一的Token异常处理机制
8. **日志记录**：记录Token相关的关键操作，便于审计和排查问题
9. **安全加固**：定期更换JWT密钥，使用HTTPS传输Token
10. **性能优化**：考虑使用Redis存储Token黑名单，提高验证性能
